create table tb_tipo(
	id_tipo serial,
	tx_descricao varchar (255),

	constraint fk_tb_tipo primary key (id_tipo)
);

create table tb_usuarioTipados (
	id_usuarioTipados serial,
	tx_nome varchar (255),
	tx_email varchar (255),
	tx_senha varchar (255),
	cd_tipo integer,

	constraint pk_tb_usuarioTipados primary key (id_usuarioTipados),
	constraint fk_tb_tipo foreign key (cd_tipo) references tb_tipo (id_tipo)
);
