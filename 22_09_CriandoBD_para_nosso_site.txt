create table tb_tipoUser(
id_tipoUser serial primary key,
tx_descricao varchar(255)
);

create table tb_usuario (
id_usuario serial primary key,
cd_tipoUser integer references tb_tipoUser (id_tipoUser),
tx_nome varchar (255) not null,
tx_userName varchar (255) unique not null,
tx_senha varchar(255) not null,
tx_email varchar (255),
nr_telefone numeric (11,0),
dt_criacao timestamp default now()
);

create table tb_animal (
id_animal serial primary key,
tx_descricao varchar (255)
);

create table tb_pet (
id_pet serial primary key,
cd_usuario integer references tb_usuario (id_usuario),
cd_animal integer references tb_animal (id_animal),
tx_raca varchar (255),
tx_nome varchar (255) not null,
tx_sexo varchar (1) not null,
vl_idade numeric (3,2), 
vl_pesoKg numeric (3,3)
);

create table tb_categoriaProduto (
id_categoriaProduto serial primary key,
tx_categoria varchar (255)
);

create table tb_produto (
id_produto serial primary key,
cd_animal integer references tb_animal(id_animal) not null,
cd_categoria integer references tb_categoriaProduto(id_categoriaProduto) not null,
tx_nome varchar (255) not null,
tx_dscricao varchar(255) not null,
vl_valor numeric (5,2) not null,
nr_estoque integer
);

create table tb_servico (
id_servico serial primary key,
tx_servico varchar (255),
vl_valor numeric (5,2),
cd_pet integer references tb_pet (id_pet),
cd_usuario integer references tb_usuario (id_usuario)
);

create table tb_venda (
id_venda serial primary key,
cd_usuario integer references tb_usuario (id_usuario),
cd_produto integer references tb_produto(id_produto),
cd_servico integer references tb_servico (id_servico),
vl_total numeric (5,2),
nr_quantidade integer,
dt_compra timestamp default now()
);
