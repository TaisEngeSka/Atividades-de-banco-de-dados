 -- buscar nome do produto, maior e menor preço dele no período de 01/02/24 e 30/03/24
select * from tb_produto 

with vendas as (
	select 
		min(historico.vl_preco_novo) as min_valor,
		max(historico.vl_preco_anterior) as max_valor,
		cd_produto
	from 
		tb_preco_historico historico
	where 
		historico.dt_alteracao 
	between  
		'2024-02-01 00:00:00.000' and '2024-03-30 00:00:00.000'
	group by cd_produto 
	)
select 
	tx_nome as nome_produto ,
	vendas.min_valor,
	vendas.max_valor
from 
	tb_produto produto 
inner join 
	vendas 
on 
	vendas.cd_produto = produto.id_produto;
