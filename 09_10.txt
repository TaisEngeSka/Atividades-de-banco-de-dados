create trigger nome
	before ou after 
	update, insert ou delete 
		on tb_atividade 
for each  -- repetir 
	row -- linha a linha
	 ou statement -- uma vez 
execute function nome_funcao(); 


returns trigger -- obrigatório quando é um trigger que chama 
current_query -- comando atual

create or replace function atividade_log_update()
returns trigger
language plpgsql
as $$
	begin
	insert into tb_atividade_log
	(tx_query, nr_idade, tx_mensagem, cd_atividade)
	values(
		current_query(),
		extract (year from age(
				old.dt_nascimento	
				)),
		'MENSAGEM VIA TRIGGER',
		new.id_atividade
	);
	return new;
	end;
$$;

create trigger nome
	before
	update on tb_atividade
for each row 
execute function atividade_log_update();

create sequence seq_produtos_log start 1 increment 1;

create table tb_pagamentos_log (
	id_pagamentos_log numeric (18,0)
		default nextval ('seq_produtos_log') primary key,
	tx_query varchar (800),
	tx_forma varchar (255),
	vl_valor_antes numeric (10,2),
	vl_valor_depois numeric (10,2),
	dt_modificacao date default current_date
);

create or replace function fn_pagamentos()
returns trigger 
language plpgsql
as $$ 
	begin
		insert into tb_pagamentos_log(
				tx_query, tx_forma, vl_valor_antes, vl_valor_depois)
		values(
				current_query(),
				lower(new.tx_forma),
				old.vl_pagamento,
				new.vl_pagamento
			);		
		return new;		
	end;
$$;

create trigger  pagamentos_log
	before insert on tb_pagamentos 
for each row 
execute function fn_pagamentos();
