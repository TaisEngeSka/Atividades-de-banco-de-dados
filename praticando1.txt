select 
	cliente.tx_nome,
	endereco.tx_cidade,
	endereco.tx_estado,
	email.tx_contato as email,
	celular.tx_contato as celular,
	telefone.tx_contato as telefone
from 
	tb_cliente cliente

	on endereco.id_endereco = cliente.cd_endereco
left join tb_contato email 
	on email.cd_cliente = cliente.id_cliente and email.cd_tipo_contato = 1
left join tb_contato celular 
	on celular.cd_cliente = cliente.id_cliente and celular.cd_tipo_contato = 2
left join tb_contato telefone 
	on telefone.cd_cliente = cliente.id_cliente and telefone.cd_tipo_contato = 3

cd_cliente
from
tb_venda
where upper(tx_status) = upper('PAGO')
group by cd_cliente
)
select
*
from
tb_venda
inner join vendamaior
on vendamaior.cd_cliente = tb_venda.cd_cliente
and tb_venda.dt_venda >= vendamaior.maior

with receitaBruta as (
		select 
			sum(nr_quantidade * vl_unitario) as receita,
			cd_empresa
		from 
			tb_item_venda item
		inner join tb_venda venda 
			on venda.id_venda = item.cd_venda 
		group by cd_empresa 
	),
	 quantidadeVendas as (
		select 
			count(cd_empresa) as quantidade,
			cd_empresa
		from 
			tb_venda venda
		where upper(venda.tx_status) = upper ('pago') 
		group by venda.cd_empresa 
	)
select 
	tx_razao_social,
	receitaBruta.receita
from 
	tb_empresa empresa
inner join quantidadeVendas 
	on quantidadeVendas.cd_empresa = empresa.id_empresa
inner join receitaBruta 
	on receitaBruta.cd_empresa = empresa.id_empresa
where 
	quantidade > 3 
	and 
	receitaBruta.receita > 1000
order by receitaBruta.receita desc

with vendas as (
select
	cd_cliente,
	sum(item.nr_quantidade * item.vl_unitario ) as valor
from 
	tb_venda
inner join tb_item_venda item 
	on item.cd_venda = id_venda
where
	upper(tx_status) = 'PAGO'
group by
	cd_cliente
), media as (
select
	sum(valor) / count(distinct cd_cliente) as mediaa 
from 
	vendas
)
select 
	* 
from 
	tb_cliente
inner join vendas on cd_cliente = id_cliente
inner join media on valor > mediaa

with media as (
select 
	  sum(nr_estoque) / count (nr_estoque) as media,
	  sum(nr_estoque) as estoque,
	  cd_categoria
from 
	tb_produto  
group by
	cd_categoria
)
select 
	produto.tx_nome as produto, 
	categoria.tx_nome as categoria,
	produto.nr_estoque as estoque_atual,
	media as media_da_categoria,
	produto.nr_estoque - media.media as quantidade_abaixo
from 
	tb_produto produto
inner join media  
	on produto.nr_estoque  < media.media 
	and media.cd_categoria = produto.cd_categoria 
inner join tb_categoria categoria 
	 on categoria.id_categoria = produto.cd_categoria;

create table atividade
as (
with compras as (
select 
	COUNT(distinct id_venda) as total_compras,
	SUM(item.nr_quantidade * item.vl_unitario) as total_gasto,
	 MIN(venda.dt_venda) as primeira_compra,
	 MAX(venda.dt_venda) as ultima_compra,
	cd_cliente
from 
	tb_venda venda
inner join tb_item_venda item 
	on item.cd_venda = venda.id_venda
where tx_status = 'PAGO' 
group by cd_cliente
)
select 
	cliente.tx_nome as nome,
	endereco.tx_cidade as cidade,
	compras.total_compras,
	compras.total_gasto,
	compras.primeira_compra,
	compras.ultima_compra,
	CASE
        WHEN total_compras < 1  THEN 'CLIENTE PEQUENO'
        WHEN total_compras = 1 THEN 'CLIENTE MEDIO'
        WHEN total_compras = 2 THEN 'CLIENTE GRANDE'
        ELSE                      'OUTRO'
    END AS status_cliente,
	(
	select
		produto.tx_nome
	from
		tb_produto produto 
	inner join tb_item_venda item on item.cd_produto = produto.id_produto 
	inner join tb_venda venda on venda.id_venda = item.cd_venda 
	where
		venda.tx_status = 'PAGO' and venda.cd_cliente  = cliente.id_cliente
		group by produto.id_produto, produto.tx_nome
	order by sum(item.nr_quantidade ) desc
	limit 1
		)
from 
	tb_cliente cliente 
inner join 
	tb_endereco endereco 
	on cliente.cd_endereco  = endereco.id_endereco 
inner join 
	compras 
	on compras.cd_cliente = cliente.id_cliente 
);
