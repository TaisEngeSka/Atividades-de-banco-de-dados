select 
	cliente.tx_nome,
	endereco.tx_cidade,
	endereco.tx_estado,
	email.tx_contato as email,
	celular.tx_contato as celular,
	telefone.tx_contato as telefone
from 
	tb_cliente cliente 
left join tb_endereco endereco 
	on endereco.id_endereco = cliente.cd_endereco
left join tb_contato email 
	on email.cd_cliente = cliente.id_cliente and email.cd_tipo_contato = 1
left join tb_contato celular 
	on celular.cd_cliente = cliente.id_cliente and celular.cd_tipo_contato = 2
left join tb_contato telefone 
	on telefone.cd_cliente = cliente.id_cliente and telefone.cd_tipo_contato = 3

select 
	cliente.tx_nome,
	email.tx_contato as email
from 
	tb_cliente cliente 
left join tb_contato email 
	on email.cd_cliente = cliente.id_cliente and email.cd_tipo_contato = 1 and email.fl_principal = true
order by cliente.tx_nome asc;

	produto.tx_descricao,
	avaliacao.nr_nota 
from

	on cliente.id_cliente = avaliacao.cd_cliente
inner join tb_produto produto 
	on avaliacao.cd_produto = produto.id_produto;

with vendamaior as (
select
max(dt_venda) as maior,
cd_cliente
from
tb_venda
where upper(tx_status) = upper('PAGO')
group by cd_cliente
)
select
*
from
tb_venda
inner join vendamaior
on vendamaior.cd_cliente = tb_venda.cd_cliente
and tb_venda.dt_venda >= vendamaior.maior

with receitaBruta as (
		select 
			sum(nr_quantidade * vl_unitario) as receita,
			cd_empresa
		from 
			tb_item_venda item
		inner join tb_venda venda 
			on venda.id_venda = item.cd_venda 
		group by cd_empresa 
	),
	 quantidadeVendas as (
		select 
			count(cd_empresa) as quantidade,
			cd_empresa
		from 
			tb_venda venda
		where upper(venda.tx_status) = upper ('pago') 
		group by venda.cd_empresa 
	)
select 
	tx_razao_social,
	receitaBruta.receita
from 
	tb_empresa empresa
inner join quantidadeVendas 
	on quantidadeVendas.cd_empresa = empresa.id_empresa
inner join receitaBruta 
	on receitaBruta.cd_empresa = empresa.id_empresa
where 
	quantidade > 3 
	and 
	receitaBruta.receita > 1000
order by receitaBruta.receita desc
