-- 6 
 select 
	tc.tx_nome, 
	count(tp.cd_categoria) as quantidade  
from  
	tb_produto tp 
inner join  
	tb_categoria tc 
	on tc.id_categoria = tp.cd_categoria and tp.fl_ativo = true 
group by 
	tc.tx_nome

-- 7
select 
	tc.tx_nome, 
	count(tp.cd_categoria) as quantidade  
from  
	tb_produto tp 
inner join  
	tb_categoria tc 
	on tc.id_categoria = tp.cd_categoria 
group by 
	tc.tx_nome  
having 
	count(tp.cd_categoria) > 2;

-- WHERE → filtra os registros antes do GROUP BY.

-- HAVING → filtra os resultados depois do GROUP BY, 
-- podendo usar COUNT(), SUM(), AVG(), etc.
