select 
	categoria.tx_nome,
	max(produto.vl_preco) as maximo,
	min (produto.vl_preco) as minimo
from 
	tb_categoria categoria 
inner join 
	tb_produto produto 
on produto.cd_categoria = categoria.id_categoria 
group by categoria.tx_nome;

select tx_nome from tb_categoria;

-- 2 

select 
	prod.tx_nome, 
	sum(pedi.nr_quantidade * prod.vl_preco) as total
from 
	tb_produto prod
inner join 
	tb_pedido pedi
on 
	pedi.cd_produto = prod.id_produto 
group by 
	prod.tx_nome
order by 
	total desc ;
