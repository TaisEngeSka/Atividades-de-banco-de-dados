-- role sem login: age como grupo de permissões
CREATE ROLE perfil_leitura;
CREATE ROLE perfil_operador;

-- role com login: usuário real que conecta ao banco
CREATE ROLE ana WITH LOGIN PASSWORD 'senha_ana';
CREATE ROLE carlos WITH LOGIN PASSWORD 'senha_carlos';

-- adicionar usuários aos grupos (herança de permissões)
GRANT perfil_leitura TO ana;
GRANT perfil_operador TO carlos;

-- remover usuário de um grupo
REVOKE perfil_leitura FROM ana;
