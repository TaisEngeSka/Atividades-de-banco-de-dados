select 
	prod.tx_nome as nomeProduto,
	prod.vl_preco as valorProduto,
	fo.tx_nome as nomeFornecedor,
	cat.tx_descricao as descricaoProduto 
from 
	tb_produto prod
inner join tb_fornecedor fo
	on fo.id_fornecedor = prod.cd_fornecedor 
inner join tb_categoria cat
	on cat.id_categoria = prod.cd_categoria 
where prod.fl_ativo = true 
order by prod.vl_preco desc;

	cli.tx_nome
from
	tb_cliente cli
left join tb_venda ven
	on cli.id_cliente = ven.cd_cliente

	on item.cd_venda = ven.id_venda
