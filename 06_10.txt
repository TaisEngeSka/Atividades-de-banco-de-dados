create function fn_nome (valor numeric)
returns varchar 
language plpgsql
as $$

	begin 

		if(valor > 18) then
			return 'Maior de idade';		
	
		elsif (valor = 18) then 
			return 'possui 18 anos';		
		
		else 
			return 'Menor de idade'; 
		end if;			
	
		end;
$$;

select 
	age(dt_nascimento),
	extract (year from age(dt_nascimento) )
from tb_atividade; 


create function fn_idade2 (valor date)
returns varchar 
language plpgsql
as $$

	begin 

		if(extract (year from age(valor) ) > 18) then
			return 'Maior de idade';		
	
		elsif (extract (year from age(valor) ) = 18) then 
			return 'possui 18 anos';		
		
		else 
			return 'Menor de idade'; 
		end if;			
	
		end;
$$;


select 
	*,
	fn_idade2(dt_nascimento) 
from tb_atividade;



create function fn_exemplo_variavel (valor date)
returns varchar 
language plpgsql
as $$
	declare 
		idade integer;
	begin
		idade := extract (year from age(valor) );
		if(idade > 18) then
			return 'Maior de idade';		
		elsif (idade = 18) then 
			return 'possui 18 anos';			
		else 
			return 'Menor de idade'; 
		end if;			
		end;
$$;
