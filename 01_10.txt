-- 2

select 
	pess.tx_nome as usuario,
	pedi.id_pedido as codigo,
	prod.tx_nome as produto,
	item.nr_quantidade as quantidade,
	prod.vl_preco as valor_unitario
from
	tb_usuarios pess
inner join tb_pedidos pedi 
on
	pedi.cd_usuario = pess.id_usuario
inner join tb_itens_pedido item 
on
	item.cd_pedido = pedi.id_pedido
inner join tb_produtos prod
on
	prod.id_produto = item.cd_produto
where
	pess.id_usuario = 1234;

-- antes 0.05
-- depois 0.003
-- 94%

create index 
 idx_pedido_usuario
on tb_pedidos(cd_usuario)

create index
	idx_item_pedido 
on tb_itens_pedido(cd_pedido);

create index 
	idx_produtos_itens 
on tb_itens_pedido(cd_produto);

drop index 
 idx_produtos_itens
