-- =====================================================
-- tabelas
-- =====================================================

create table tb_tipouser (
    id_tipouser numeric(18,0)
        default nextval('seq_tipouser') primary key,
    tx_descricao varchar(255) not null
);

create table tb_usuario (
    id_usuario numeric(18,0)
        default nextval('seq_usuario') primary key,
    cd_tipouser numeric(18,0) not null,
    tx_nome varchar(255) not null,
    tx_username varchar(255) unique not null,
    tx_senha varchar(255) not null,
    tx_email varchar(255),
    nr_telefone numeric(11,0),
    dt_criacao timestamp default now(),
    constraint fk_usuario_tipouser
        foreign key (cd_tipouser)
        references tb_tipouser(id_tipouser)
);

create table tb_animal (
    id_animal numeric(18,0)
        default nextval('seq_animal') primary key,
    tx_descricao varchar(255) not null
);

create table tb_pet (
    id_pet numeric(18,0)
        default nextval('seq_pet') primary key,
    cd_usuario numeric(18,0) not null,
    cd_animal numeric(18,0) not null,
    tx_raca varchar(255),
    tx_nome varchar(255) not null,
    tx_sexo varchar(1) not null,
    vl_idade numeric(4,2),
    vl_pesoKg numeric(6,3),
    constraint fk_pet_usuario
        foreign key (cd_usuario)
        references tb_usuario(id_usuario),
    constraint fk_pet_animal
        foreign key (cd_animal)
        references tb_animal(id_animal),
    constraint ck_pet_sexo
        check (tx_sexo in ('m', 'f')),
    constraint ck_pet_idade
        check (vl_idade is null or vl_idade >= 0),
    constraint ck_pet_peso
        check (vl_pesoKg is null or vl_pesoKg >= 0)
);

create table tb_categoriaproduto (
    id_categoriaproduto numeric(18,0)
        default nextval('seq_categoriaproduto') primary key,
    tx_categoria varchar(255) not null
);

create table tb_produto (
    id_produto numeric(18,0)
        default nextval('seq_produto') primary key,
    cd_animal numeric(18,0) not null,
    cd_categoria numeric(18,0) not null,
    tx_nome varchar(255) not null,
    tx_dscricao varchar(255) not null,
    vl_valor numeric(10,2) not null,
    nr_estoque integer not null default 0,
    constraint fk_produto_animal
        foreign key (cd_animal)
        references tb_animal(id_animal),
    constraint fk_produto_categoria
        foreign key (cd_categoria)
        references tb_categoriaproduto(id_categoriaproduto),
    constraint ck_produto_valor
        check (vl_valor > 0),
    constraint ck_produto_estoque
        check (nr_estoque >= 0)
);

create table tb_servico (
    id_servico numeric(18,0)
        default nextval('seq_servico') primary key,
    tx_servico varchar(255) not null,
    vl_valor numeric(10,2) not null,
    cd_pet numeric(18,0) not null,
    cd_usuario numeric(18,0) not null,
    dt_compra timestamp default now(),
    constraint fk_servico_pet
        foreign key (cd_pet)
        references tb_pet(id_pet),
    constraint fk_servico_usuario
        foreign key (cd_usuario)
        references tb_usuario(id_usuario),
    constraint ck_servico_valor
        check (vl_valor > 0)
);

create table tb_venda (
    id_venda numeric(18,0)
        default nextval('seq_venda') primary key,
    cd_usuario numeric(18,0) not null,
    cd_produto numeric(18,0) not null,
    vl_total numeric(12,2) not null default 0,
    nr_quantidade integer not null,
    dt_compra timestamp default now(),
    constraint fk_venda_usuario
        foreign key (cd_usuario)
        references tb_usuario(id_usuario),
    constraint fk_venda_produto
        foreign key (cd_produto)
        references tb_produto(id_produto),
    constraint ck_venda_quantidade
        check (nr_quantidade > 0),
    constraint ck_venda_total
        check (vl_total >= 0)
);

-- function

create or replace function fn_total_compras_usuario(p_id_usuario numeric(18,0))
returns numeric(12,2)
language plpgsql
as $$
declare
    v_total numeric(12,2);
begin
    select coalesce(sum(vl_total), 0)
    into v_total
    from tb_venda
    where cd_usuario = p_id_usuario;
    return v_total;
end;
$$;

-- =====================================================
-- function da trigger
-- =====================================================

create or replace function fn_controlar_estoque()
returns trigger
language plpgsql
as $$
declare
    v_valor numeric(10,2);
    v_estoque integer;
begin
    select vl_valor, nr_estoque
    into v_valor, v_estoque
    from tb_produto
    where id_produto = new.cd_produto
    for update;

    if not found then
        raise exception 'produto % nao encontrado', new.cd_produto;
    end if;

    if v_estoque < new.nr_quantidade then
        raise exception 'estoque insuficiente. disponivel: %, solicitado: %',
            v_estoque, new.nr_quantidade;
    end if;

    new.vl_total := v_valor * new.nr_quantidade;

    update tb_produto
    set nr_estoque = nr_estoque - new.nr_quantidade
    where id_produto = new.cd_produto;

    return new;
end;
$$;

-- =====================================================
-- trigger
-- =====================================================

create trigger trg_controlar_estoque
before insert on tb_venda
for each row
execute function fn_controlar_estoque();

create or replace view vw_relatorio_vendas as
select
    ven.id_venda,
    usu.tx_nome as nome_cliente,
    pro.tx_nome as nome_produto,
    ven.nr_quantidade,
    pro.vl_valor as valor_unitario,
    ven.vl_total,
    ven.dt_compra
from tb_venda ven
inner join tb_usuario usu
    on usu.id_usuario = ven.cd_usuario
inner join tb_produto pro
    on pro.id_produto = ven.cd_produto;
